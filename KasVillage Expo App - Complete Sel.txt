// ARWEAVE DIRECT CLIENT (Phone queries Arweave directly)
// ============================================================================
const ArweaveClient = {
  // Search all KasVillage content by type
  search: async (contentType: string, query?: string): Promise<ArweaveItem[]> => {
    const gql = `
      query {
        transactions(
          tags: [
            { name: "App-Name", values: ["${APP_TAG}"] },
            { name: "Type", values: ["${contentType}"] }
          ],
          first: 100,
          sort: HEIGHT_DESC
        ) {
          edges {
            node {
              id
              owner { address }
              tags { name value }
              block { timestamp height }
            }
          }
        }
      }
    `;
    
    try {
      const res = await fetch(ARWEAVE_GRAPHQL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ query: gql }),
      });
      const json = await res.json();
      const edges = json?.data?.transactions?.edges || [];
      
      const items: ArweaveItem[] = [];
      for (const edge of edges) {
        const node = edge.node;
        const tags = node.tags.reduce((acc: any, t: any) => ({ ...acc, [t.name]: t.value }), {});
        
        // Fetch actual data
        try {
          const dataRes = await fetch(`${ARWEAVE_GATEWAY}/${node.id}`);
          const data = await dataRes.json();
          
          // Filter by query if provided
          if (query) {
            const searchStr = JSON.stringify(data).toLowerCase();
            if (!searchStr.includes(query.toLowerCase())) continue;
          }
          
          items.push({
            txId: node.id,
            ownerPubkey: tags['Owner-Pubkey'] || node.owner.address,
            type: tags['Type'],
            createdAt: parseInt(tags['Created-At'] || '0'),
            data,
          });
        } catch {
          // Skip items that fail to load
        }
      }
      
      return items;
    } catch (e) {
      console.error('Arweave search error:', e);
      return [];
    }
  },
  
  // Search stores
  searchStores: async (query?: string) => ArweaveClient.search(TAG_STORE, query),
  
  // Search coupons
  searchCoupons: async (query?: string) => ArweaveClient.search(TAG_COUPON, query),
  
  // Search DApps
  searchDApps: async (query?: string) => ArweaveClient.search(TAG_DAPP, query),
  
  // Search academics
  searchAcademics: async (query?: string) => ArweaveClient.search(TAG_ACADEMIC, query),
  
  // Search services
  searchServices: async (query?: string) => ArweaveClient.search(TAG_SERVICE, query),
  
  // Get item by ID
  getById: async (txId: string): Promise<ArweaveItem | null> => {
    try {
      const dataRes = await fetch(`${ARWEAVE_GATEWAY}/${txId}`);
      const data = await dataRes.json();
      return {
        txId,
        ownerPubkey: data.ownerPubkey || '',
        type: data.type || 'Unknown',
        createdAt: data.createdAt || 0,
        data,
      };
    } catch {
      return null;
    }
  },
};

// DIRECT ARWEAVE PUBLISHER (Phone publishes directly via Bundlr)
// All uploads include attestation hash + signature for tamper-proofing
// ============================================================================
const ArweavePublisher = {
  // Upload signed data to Arweave via Bundlr
  uploadSigned: async (
    signedData: SignedArweaveData,
    baseTags: { name: string; value: string }[],
  ): Promise<{ txId: string } | { error: string }> => {
    try {
      // Add signature tags
      const signatureTags = ArweaveSignatureManager.createSignedTags(signedData);
      const allTags = [...baseTags, ...signatureTags];
      
      const dataBytes = new TextEncoder().encode(JSON.stringify(signedData));
      
      const res = await fetch(`${BUNDLR_NODE}/tx/arweave`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/octet-stream',
          'x-tags': JSON.stringify(allTags),
        },
        body: dataBytes,
      });
      
      if (!res.ok) {
        const errText = await res.text();
        return { error: `Upload failed: ${errText}` };
      }
      
      const result = await res.json();
      return { txId: result.id };
    } catch (e: any) {
      return { error: e.message };
    }
  },
  
  // Legacy upload (without signing - kept for compatibility)
  upload: async (
    data: any, 
    tags: { name: string; value: string }[],
    _signingKey: string
  ): Promise<{ txId: string } | { error: string }> => {
    try {
      const dataBytes = new TextEncoder().encode(JSON.stringify(data));
      
      const res = await fetch(`${BUNDLR_NODE}/tx/arweave`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/octet-stream',
          'x-tags': JSON.stringify(tags),
        },
        body: dataBytes,
      });
      
      if (!res.ok) {
        const errText = await res.text();
        return { error: `Upload failed: ${errText}` };
      }
      
      const result = await res.json();
      return { txId: result.id };
    } catch (e: any) {
      return { error: e.message };
    }
  },
  
  // Publish store with attestation (FREE - small data)
  publishStore: async (
    storeData: any, 
    ownerPubkey: string, 
    attestation: AttestationResult
  ): Promise<{ txId: string } | { error: string }> => {
    const signedData = await ArweaveSignatureManager.signData(storeData, ownerPubkey, attestation);
    const tags = [
      { name: 'App-Name', value: APP_TAG },
      { name: 'Type', value: TAG_STORE },
      { name: 'Owner-Pubkey', value: ownerPubkey },
      { name: 'Created-At', value: Date.now().toString() },
      { name: 'Content-Type', value: 'application/json' },
    ];
    return ArweavePublisher.uploadSigned(signedData, tags);
  },
  
  // Publish coupon with attestation (FREE - small data)
  publishCoupon: async (
    couponData: any, 
    ownerPubkey: string, 
    attestation: AttestationResult
  ): Promise<{ txId: string } | { error: string }> => {

// ============================================================================
// DEVICE INTEGRITY MANAGER
// Detects rooted/jailbroken devices, emulators, tampered environments
// ============================================================================
interface IntegrityResult {
  compromised: boolean;
  reason: string | null;
  details?: string;
}

const DeviceIntegrityManager = {
  checkIntegrity: async (): Promise<IntegrityResult> => {
    try {
      // Check for rooted/jailbroken device
      const isRooted = await Device.isRootedExperimentalAsync();
      if (isRooted) {
        return {
          compromised: true,
          reason: 'untrusted_os_root',
          details: Platform.OS === 'ios' ? 'Jailbroken device detected' : 'Rooted device detected',
        };
      }
      
      // Check for emulator/simulator
      if (!Device.isDevice) {
        return {
          compromised: true,
          reason: 'virtual_environment',
          details: Platform.OS === 'ios' ? 'iOS Simulator detected' : 'Android Emulator detected',
        };
      }
      
      // Check for suspicious device names (common emulators)
      if (Platform.OS === 'android') {
        const deviceName = (Device.deviceName || '').toLowerCase();
        const modelName = (Device.modelName || '').toLowerCase();
        const suspicious = ['sdk', 'emulator', 'genymotion', 'bluestacks', 'nox', 'memu', 'andy'];
        for (const s of suspicious) {
          if (deviceName.includes(s) || modelName.includes(s)) {
            return {
              compromised: true,
              reason: 'suspicious_device_name',
              details: `Emulator detected: ${Device.modelName}`,
            };
          }
        }
      }
      
      return { compromised: false, reason: null };
    } catch (error: any) {
      // Fail safe - if check fails, assume not compromised
      return { compromised: false, reason: null, details: `Check failed: ${error.message}` };
    }
  },
  
  getFingerprint: () => ({
    brand: Device.brand,
    manufacturer: Device.manufacturer,
    modelName: Device.modelName,
    modelId: Device.modelId,
    osName: Device.osName,
    osVersion: Device.osVersion,
    deviceYearClass: Device.deviceYearClass,
    totalMemory: Device.totalMemory,
    isDevice: Device.isDevice,
  }),
};

// ============================================================================
// ATTESTATION MANAGER
// iOS App Attest / Android Play Integrity
// Proves app and device haven't been tampered with
// ============================================================================
interface AttestationResult {
  success: boolean;
  platform: 'ios' | 'android' | 'web';
  attestationBlob?: string;
  keyId?: string;
  nonce: string;
  timestamp: number;
  deviceIntegrity: boolean;
  appIntegrity: boolean;
  error?: string;
}

const AttestationManager = {
  keyId: null as string | null,
  lastAttestation: null as AttestationResult | null,
  
  // In production: set to true with Apple Developer / Play Console setup
  IOS_ENABLED: false,
  ANDROID_ENABLED: false,
  
  getPlatform: (): 'ios' | 'android' | 'web' => {
    if (Platform.OS === 'ios') return 'ios';
    if (Platform.OS === 'android') return 'android';
    return 'web';
  },
  
  isAvailable: async (): Promise<boolean> => {
    const platform = AttestationManager.getPlatform();
    if (platform === 'ios') return AttestationManager.IOS_ENABLED;
    if (platform === 'android') return AttestationManager.ANDROID_ENABLED;
    return false;
  },
  
  // Perform device attestation with server challenge
  performAttestation: async (challenge: string): Promise<AttestationResult> => {
    const platform = AttestationManager.getPlatform();
    const timestamp = Date.now();
    
    // First check device integrity
    const integrity = await DeviceIntegrityManager.checkIntegrity();
    if (integrity.compromised) {
      return {
        success: false,
        platform,
        nonce: challenge,
        timestamp,
        deviceIntegrity: false,
        appIntegrity: false,
        error: integrity.details || integrity.reason || 'Device compromised',
      };
    }
    
    if (platform === 'ios') {
      // Mock iOS App Attest (real impl requires Apple Developer account)
      // In production: use react-native-app-attest
      const mockBlob = await Crypto.digestStringAsync(
        Crypto.CryptoDigestAlgorithm.SHA256,
        `ios_attestation_${challenge}_${timestamp}_${Device.modelId}`
      );
      const keyId = `ios_key_${timestamp}`;
      AttestationManager.keyId = keyId;
      await SecureStore.setItemAsync('attestation_key_id', keyId);
      
      const result: AttestationResult = {
        success: true,
        platform: 'ios',
        attestationBlob: `IOS_ATTESTATION_${mockBlob}`,
        keyId,
        nonce: challenge,
        timestamp,
        deviceIntegrity: true,
        appIntegrity: true,
      };
      AttestationManager.lastAttestation = result;
      return result;
    }
    
    if (platform === 'android') {
      // Mock Play Integrity (real impl requires Play Console)
      const mockToken = await Crypto.digestStringAsync(
        Crypto.CryptoDigestAlgorithm.SHA256,
        `android_integrity_${challenge}_${timestamp}_${Device.modelId}`
      );
      
      const result: AttestationResult = {
        success: true,
        platform: 'android',
        attestationBlob: `ANDROID_TOKEN_${mockToken}`,
        nonce: challenge,
        timestamp,
        deviceIntegrity: true,
        appIntegrity: true,
      };
      AttestationManager.lastAttestation = result;
      return result;
    }
    
    // Web - no hardware attestation available
    return {
      success: true,
      platform: 'web',
      nonce: challenge,
      timestamp,
      deviceIntegrity: false,
      appIntegrity: false,
    };
  },
  
  // Compute attestation hash for L1 inscription (16 bytes)
  computeAttestationHash: async (attestation: AttestationResult): Promise<string> => {
    const data = `${attestation.attestationBlob || ''}_${attestation.nonce}_${attestation.timestamp}_${attestation.keyId || ''}`;
    const fullHash = await Crypto.digestStringAsync(Crypto.CryptoDigestAlgorithm.SHA256, data);
    return fullHash.slice(0, 32); // First 16 bytes = 32 hex chars
  },
  
  getLastAttestation: (): AttestationResult | null => AttestationManager.lastAttestation,
};

// ============================================================================
// ARWEAVE SIGNATURE MANAGER
// Signs data before publishing to prove it came from untampered app
// ============================================================================
interface SignedArweaveData {
  data: any;
  signature: string;
  signerPubkey: string;
  attestationHash: string;
  timestamp: number;
}

const ArweaveSignatureManager = {
  // Sign data with user's key + attestation binding
  signData: async (
    data: any,
    publicKeyHex: string,
    attestation: AttestationResult,
  ): Promise<SignedArweaveData> => {
    const timestamp = Date.now();
    
    // Get attestation hash
    const attestationHash = await AttestationManager.computeAttestationHash(attestation);
    
    // Create message: hash of (data + attestation + timestamp)
    const dataJson = JSON.stringify(data);
    const messageToSign = await Crypto.digestStringAsync(
      Crypto.CryptoDigestAlgorithm.SHA256,
      `${dataJson}_${attestationHash}_${timestamp}`
    );
    
    // Sign with secp256k1 (in production: use crypto library)
    // For now: HMAC-based signature placeholder
    const signature = await Crypto.digestStringAsync(
      Crypto.CryptoDigestAlgorithm.SHA256,
      `SIGN_${messageToSign}_${publicKeyHex}`
    );
    
    return {
      data,
      signature,
      signerPubkey: publicKeyHex,
      attestationHash,
      timestamp,
    };
  },
  
  // Verify signed data matches expected pubkey
  verifySignature: (signedData: SignedArweaveData, expectedPubkey: string): boolean => {
    // In production: verify secp256k1 ECDSA signature
    return signedData.signerPubkey === expectedPubkey && 
           signedData.signature.length > 0 &&
           signedData.attestationHash.length > 0;
  },
  
  // Create signed tags for Arweave upload
  createSignedTags: (signedData: SignedArweaveData): { name: string; value: string }[] => [
    { name: 'Signer-Pubkey', value: signedData.signerPubkey },
    { name: 'Attestation-Hash', value: signedData.attestationHash },
    { name: 'Signature', value: signedData.signature },
    { name: 'Signed-At', value: signedData.timestamp.toString() },
  ],
};

const HumanVerifier = {
  verify: async (): Promise<{ verified: boolean; attestation?: string }> => {
    try {
      const result = await LocalAuthentication.authenticateAsync({
        promptMessage: 'Verify you are human',
        disableDeviceFallback: true,
      });
      if (!result.success) return { verified: false };
      
      const attestation = await Crypto.digestStringAsync(
        Crypto.CryptoDigestAlgorithm.SHA256,
        `human_${Date.now()}_${Device.modelId}`
      );
      return { verified: true, attestation };
    } catch {
      return { verified: false };
    }
  },
};

// SEARCH SCREEN (Queries Arweave directly from phone)
// ============================================================================
const SearchScreen: React.FC = () => {
  const [query, setQuery] = useState('');
  const [searchType, setSearchType] = useState<'stores' | 'coupons' | 'dapps' | 'academics' | 'services'>('coupons');
  const [results, setResults] = useState<ArweaveItem[]>([]);
  const [loading, setLoading] = useState(false);
  const [selectedItem, setSelectedItem] = useState<ArweaveItem | null>(null);
  const [verificationResult, setVerificationResult] = useState<any>(null);

  const doSearch = async () => {
    Keyboard.dismiss();
    setLoading(true);
    setResults([]);
    
    let items: ArweaveItem[] = [];
    switch (searchType) {
      case 'stores': items = await ArweaveClient.searchStores(query); break;
      case 'coupons': items = await ArweaveClient.searchCoupons(query); break;
      case 'dapps': items = await ArweaveClient.searchDApps(query); break;
      case 'academics': items = await ArweaveClient.searchAcademics(query); break;
      case 'services': items = await ArweaveClient.searchServices(query); break;
    }
    
    // Sort by price (cheapest first) then KAS discount
    items.sort((a, b) => {
      const priceA = a.data?.dollarPrice || a.data?.price || 0;
      const priceB = b.data?.dollarPrice || b.data?.price || 0;
      if (priceA !== priceB) return priceA - priceB;
      const discountA = a.data?.kasDiscountPercent || 0;
      const discountB = b.data?.kasDiscountPercent || 0;
      return discountB - discountA;
    });
    
    setResults(items);
    setLoading(false);
  };

  const verifyItem = async (item: ArweaveItem) => {
    setSelectedItem(item);
    const type = item.type === 'Store' ? 'store' : item.type === 'DApp' ? 'dapp' : 'user';
    const result = await TownHallVerifier.getStatus(item.txId, type as any);
    setVerificationResult(result);
  };

  const renderItem = ({ item }: { item: ArweaveItem }) => (
    <TouchableOpacity style={styles.resultCard} onPress={() => verifyItem(item)}>
      <View style={styles.resultHeader}>
        <Text style={styles.resultTitle}>{item.data?.name || item.data?.displayText || 'Unnamed'}</Text>
        {item.verified && <Text style={styles.verifiedBadge}>✓ Verified</Text>}
      </View>
      <Text style={styles.resultDesc} numberOfLines={2}>{item.data?.description || ''}</Text>
      <View style={styles.resultFooter}>
        {item.data?.dollarPrice && (
          <Text style={styles.priceText}>${item.data.dollarPrice.toFixed(2)}</Text>
        )}
        {item.data?.kasDiscountPercent > 0 && (
          <Text style={styles.discountText}>{item.data.kasDiscountPercent}% KAS discount</Text>
        )}
      </View>
    </TouchableOpacity>
  );

  return (
    <View style={styles.screen}>
      <View style={styles.searchBar}>
        <TextInput
          style={styles.searchInput}
          placeholder="Search marketplace..."
          value={query}
          onChangeText={setQuery}
          onSubmitEditing={doSearch}
          returnKeyType="search"
        />
        <TouchableOpacity style={styles.searchButton} onPress={doSearch}>
          <Text style={styles.searchButtonText}>🔍</Text>
        </TouchableOpacity>
      </View>
      
      <ScrollView horizontal style={styles.typeSelector} showsHorizontalScrollIndicator={false}>
        {(['coupons', 'stores', 'dapps', 'academics', 'services'] as const).map(type => (
          <TouchableOpacity
            key={type}
            style={[styles.typeButton, searchType === type && styles.typeButtonActive]}
            onPress={() => setSearchType(type)}
          >
            <Text style={[styles.typeButtonText, searchType === type && styles.typeButtonTextActive]}>
              {type.charAt(0).toUpperCase() + type.slice(1)}
            </Text>
          </TouchableOpacity>
        ))}
      </ScrollView>

      {loading ? (
        <ActivityIndicator size="large" style={styles.loader} />
      ) : (
        <FlatList
          data={results}
          renderItem={renderItem}
          keyExtractor={item => item.txId}
          contentContainerStyle={styles.resultsList}
          ListEmptyComponent={
            <Text style={styles.emptyText}>
              {query ? 'No results found' : 'Enter a search term'}
            </Text>
          }
        />
      )}

      {/* Verification Modal */}
      <Modal visible={!!selectedItem} animationType="slide" transparent>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>{selectedItem?.data?.name || 'Item Details'}</Text>
            
            {verificationResult ? (
              <View style={styles.verificationBox}>
                <Text style={styles.verificationStatus}>
                  {verificationResult.verified ? '✓ VERIFIED' : '⚠ NOT VERIFIED'}
                </Text>
                {verificationResult.stats && (
                  <View style={styles.statsBox}>
                    <Text>XP: {verificationResult.stats.xp}</Text>
                    <Text>Completion: {(verificationResult.stats.pComplete * 100).toFixed(0)}%</Text>
                    <Text>Deadlocks: {verificationResult.stats.deadlocks}</Text>
                  </View>
                )}
                {verificationResult.proof && (
                  <Text style={styles.proofText}>Proof: {verificationResult.proof.slice(0, 20)}...</Text>
                )}
              </View>
            ) : (
              <ActivityIndicator />
            )}
            
            <TouchableOpacity 
              style={styles.closeButton} 
              onPress={() => { setSelectedItem(null); setVerificationResult(null); }}
            >
              <Text style={styles.closeButtonText}>Close</Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>
    </View>
  );
};

// TOWN HALL SCREEN (Verification only - the ONLY backend we run)
// ============================================================================
const TownHallScreen: React.FC = () => {
  const [inputId, setInputId] = useState('');
  const [idType, setIdType] = useState<'apt' | 'store' | 'dapp'>('apt');
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<any>(null);

  const verify = async () => {
    if (!inputId.trim()) return Alert.alert('Error', 'Enter an ID to verify');
    
    Keyboard.dismiss();
    setLoading(true);
    setResult(null);
    
    let verifyResult: any;
    switch (idType) {
      case 'apt':
        verifyResult = await TownHallVerifier.verifyUser(inputId);
        break;
      case 'store':
        verifyResult = await TownHallVerifier.verifyStore(inputId);
        break;
      case 'dapp':
        verifyResult = await TownHallVerifier.verifyDApp(inputId);
        break;
    }
    
    setResult(verifyResult);

    (function() {
      'use strict';
      const BLOCKED = 'KasVillage: Blocked';
      Object.defineProperty(window, 'localStorage', { get: () => { throw new Error(BLOCKED); } });
      Object.defineProperty(window, 'sessionStorage', { get: () => { throw new Error(BLOCKED); } });
      window.open = () => { throw new Error(BLOCKED); };
      window.eval = () => { throw new Error(BLOCKED); };
      
      window.kasvillageSDK = {
        getPublicKey: () => window.ReactNativeWebView.postMessage(JSON.stringify({ type: 'GET_PUBKEY' })),
        requestPayment: (amount, recipient, memo) => {
          if (amount > 100000000000) throw new Error('Exceeds $1000 cap');
          window.ReactNativeWebView.postMessage(JSON.stringify({ type: 'PAYMENT', amount, recipient, memo }));
        },
        version: '1.0.0'
      };
      true;
    })();

// MAIN APP
// ============================================================================
const App: React.FC = () => {
  const [activeTab, setActiveTab] = useState<'wallet' | 'search' | 'workspace' | 'townhall'>('wallet');
  const [wallet, setWallet] = useState<WalletState>({
    initialized: false,
    publicKey: null,
    kaspaAddress: null,
    aptAlias: null,
    authenticated: false,
    xp: 0,
    pComplete: 1.0,
    deadlocks: 0,
    inSnailMode: false,
  });
  const [blocked, setBlocked] = useState(false);
  const [blockReason, setBlockReason] = useState('');

  useEffect(() => {
    initializeApp();
  }, []);

  const initializeApp = async () => {
    // Geo check
    const geo = await GeoChecker.check();
    if (!geo.allowed) {
      setBlocked(true);
      setBlockReason(geo.reason || 'Service unavailable');
      return;
    }
    
    // Load wallet
    const publicKey = await SecureStore.getItemAsync('public_key');
    const kaspaAddress = await SecureStore.getItemAsync('kaspa_address');
    const aptAlias = await SecureStore.getItemAsync('apt_alias');
    
    if (publicKey) {
      setWallet(prev => ({
        ...prev,
        initialized: true,
        publicKey,
        kaspaAddress,
        aptAlias,
      }));
    } else {
      // Generate new wallet
      const keyBytes = await Crypto.getRandomBytesAsync(32);
      const newPubKey = Array.from(keyBytes).map(b => b.toString(16).padStart(2, '0')).join('');
      await SecureStore.setItemAsync('public_key', newPubKey);
      
      // Generate APT alias
      const hash = await Crypto.digestStringAsync(Crypto.CryptoDigestAlgorithm.SHA256, newPubKey);
      const aptNum = parseInt(hash.slice(0, 4), 16) % 1000;
      const newAptAlias = `APT-${aptNum.toString().padStart(3, '0')}`;
      await SecureStore.setItemAsync('apt_alias', newAptAlias);
      
      setWallet(prev => ({
        ...prev,
        initialized: true,
        publicKey: newPubKey,
        aptAlias: newAptAlias,
      }));
    }
  };

  const refreshWallet = async () => {
    // Fetch latest stats from Town Hall
    if (wallet.aptAlias) {
      const result = await TownHallVerifier.verifyUser(wallet.aptAlias);
      if (result.stats) {
        setWallet(prev => ({
          ...prev,
          xp: result.stats.xp,
          pComplete: result.stats.pComplete,
          deadlocks: result.stats.deadlocks,
          inSnailMode: result.stats.snailMode,
        }));
      }
    }
  };

  if (blocked) {
    return (
      <View style={styles.blockedScreen}>
        <Text style={styles.blockedTitle}>⛔ Access Denied</Text>
        <Text style={styles.blockedText}>{blockReason}</Text>
      </View>
    );
  }

  if (!wallet.initialized) {
    return (
      <View style={styles.loadingScreen}>
        <ActivityIndicator size="large" color="#49EACB" />
        <Text style={styles.loadingText}>Initializing...</Text>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      {/* Content */}
      <View style={styles.content}>
        {activeTab === 'wallet' && <WalletScreen wallet={wallet} onRefresh={refreshWallet} />}
        {activeTab === 'search' && <SearchScreen />}
        {activeTab === 'workspace' && <WorkspaceScreen wallet={wallet} />}
        {activeTab === 'townhall' && <TownHallScreen />}
      </View>
      
      {/* Tab Bar */}
      <View style={styles.tabBar}>
        {[
          { id: 'wallet', icon: '💳', label: 'Wallet' },
          { id: 'search', icon: '🔍', label: 'Search' },
          { id: 'workspace', icon: '📦', label: 'Workspace' },
          { id: 'townhall', icon: '🏛️', label: 'Town Hall' },
        ].map(tab => (
          <TouchableOpacity
            key={tab.id}
            style={[styles.tabItem, activeTab === tab.id && styles.tabItemActive]}
            onPress={() => setActiveTab(tab.id as any)}
          >
            <Text style={styles.tabIcon}>{tab.icon}</Text>
            <Text style={[styles.tabLabel, activeTab === tab.id && styles.tabLabelActive]}>
              {tab.label}
            </Text>
          </TouchableOpacity>
        ))}
      </View>
    </View>
  );
};
